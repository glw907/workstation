# Curated GNOME settings: each line is one deliberate deviation from the
# stock Bluefin/GNOME defaults, as "schema key value", where value is the
# serialized form `gsettings get` prints. bootstrap.sh applies the lines;
# check-drift compares them against the live dconf database. Keep this
# file curated by hand; never regenerate it from `dconf dump`, which
# captures transient state (window sizes, recent files) alongside intent.

# Stock GNOME suspends after 15 idle minutes even on AC power. Battery
# keeps the stock suspend behavior on purpose.
org.gnome.settings-daemon.plugins.power sleep-inactive-ac-type 'nothing'

# A calm, low-distraction desktop (Geoff, 2026-09-29). Do Not Disturb hides
# notification banners, but GNOME Shell still shows critical-urgency ones, so
# only critical items interrupt. Nothing shows on the lock screen. Missed
# notifications still collect in the calendar drop-down.
org.gnome.desktop.notifications show-banners false
org.gnome.desktop.notifications show-in-lock-screen false

# The dock shows no unread counts on app icons (Discord, Hylki, and the
# rest), and no icon wiggles to demand attention.
org.gnome.shell.extensions.dash-to-dock show-icons-emblems false
org.gnome.shell.extensions.dash-to-dock show-icons-notifications-counter false
org.gnome.shell.extensions.dash-to-dock dance-urgent-applications false

# The dock, left to right (Geoff, 2026-10-13, from thinkpad-x1; Calendar and
# Contacts added 2026-10-21). Trash sits
# after these through dash-to-dock's stock show-trash.
org.gnome.shell favorite-apps ['org.gnome.Ptyxis.desktop', 'org.gnome.Nautilus.desktop', 'chromium-browser.desktop', 'co.hyprlab.Hylki.desktop', 'org.gnome.Calendar.desktop', 'org.gnome.Contacts.desktop', 'com.discordapp.Discord.desktop', 'io.github.janbar.noson.desktop', 'io.github.screwys.Rufin.desktop', 'com.valvesoftware.Steam.desktop', 'com.belmoussaoui.Authenticator.desktop']

# Super+Shift+P opens phone-check, which answers a site's phone check through
# the voip.ms number (Geoff has no mobile phone; 2026-10-29). The list keeps
# Bluefin's six stock shortcuts, custom0 to custom6, and adds custom7 at the end.
org.gnome.settings-daemon.plugins.media-keys custom-keybindings ['/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom0/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom1/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom2/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom3/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom5/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom6/', '/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom7/']
org.gnome.settings-daemon.plugins.media-keys.custom-keybinding:/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom7/ name 'Phone check'
org.gnome.settings-daemon.plugins.media-keys.custom-keybinding:/org/gnome/settings-daemon/plugins/media-keys/custom-keybindings/custom7/ command 'ptyxis --new-window -- bash -lc "phone-check --pause"'
